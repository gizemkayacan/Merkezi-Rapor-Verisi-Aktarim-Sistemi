namespace DynamicExcelImport.API.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ImportGroupController : ControllerBase
    {
        private readonly DatabaseService _dbService;

        public ImportGroupController(DatabaseService dbService)
        {
            _dbService = dbService;
        }

        // 1. HTTP GET - Alt Grupları Listeleme
        [HttpGet] // Bu metodun sadece dışarıdan "Bana veri getir" (GET) isteklerine cevap vereceğini söyler.
        public IActionResult GetGroups()
        {
            var groups = new List<object>(); // Veritabanından gelecek satırları dolduracağımız boş bir liste (sepet) yaratıyoruz.

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open(); // SQL Server bağlantısını fiilen başlatıyoruz.
                // Dikkat: Bu sefer depodan "UpperGroupId" (Sahibinin kimliği) bilgisini de çekiyoruz
                string query = "SELECT Id, UpperGroupId, Name, Code FROM dbo.ImportGroup WHERE IsActive = 1"; // Çalıştıracağımız SQL sorgusu. Dikkat: Sadece IsActive=1 (Silinmemiş, aktif) olanları getiriyoruz.

                using (var command = new SqlCommand(query, (SqlConnection)connection)) // Sorguyu ve açık olan kapıyı bir komut paketine dönüştürüyoruz.
                {
                    using (var reader = command.ExecuteReader()) // Komutu çalıştırıyor ve gelen sonuçları okumaya (reader) başlıyoruz.
                    {
                        while (reader.Read()) // Gelen sonuçlarda okunacak satır olduğu sürece bu döngü döner (Örn: 5 kayıt varsa 5 kere döner)
                        {
                            groups.Add(new // Boş sepete yeni bir obje ekliyoruz.
                            {
                                Id = reader.GetInt64(0), // 0. sıradaki kolonu (Id) sayı olarak oku.
                                UpperGroupId = reader.GetInt64(1),
                                Name = reader.GetString(2),
                                Code = reader.GetString(3)
                            });
                        }
                    }
                }
            }
            return Ok(groups);
        }

        // 2. HTTP POST - Sisteme Yeni Alt Grup Ekleme (İlişkisel Veri)
        [HttpPost] // Bu metodun sadece sisteme "Yeni veri ekle" (POST) isteklerinde çalışacağını söyler.
        public IActionResult CreateGroup([FromBody] CreateGroupRequest request) // Gelen JSON verisini en alttaki CreateGroupRequest şablonuna oturtup 'request' adıyla içeri alır.
        {
            // Temel boşluk kontrolü
            if (string.IsNullOrWhiteSpace(request.Name) || string.IsNullOrWhiteSpace(request.Code))
            {
                return BadRequest("Alt grup adı ve kodu boş olamaz!"); // İsim veya kod boşsa işlemi reddet (HTTP 400).
            }

            if (request.UpperGroupId <= 0)
            {
                return BadRequest("Geçerli bir Ana Klasör (UpperGroup) ID'si girmelisiniz!"); // ID 0 veya eksiyse reddet.
            }

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open(); // SQL kapısını açtık.

                // --- GÜVENLİK POLİSİ DEVREDE ---
                // Kullanıcının gönderdiği UpperGroupId gerçekten veritabanında var mı diye kontrol ediyoruz
                string checkQuery = "SELECT COUNT(1) FROM dbo.ImportUpperGroup WHERE Id = @UpperGroupId AND IsActive = 1";
                using (var checkCommand = new SqlCommand(checkQuery, (SqlConnection)connection))
                {
                    checkCommand.Parameters.AddWithValue("@UpperGroupId", request.UpperGroupId); // Güvenlik: ID'yi parametre olarak gönderiyoruz (SQL Injection engeli).

                    // ExecuteScalar tek bir değer (burada sayı) dönen sorgular için kullanılır
                    int upperGroupExists = Convert.ToInt32(checkCommand.ExecuteScalar()); // ExecuteScalar: Sadece tek bir hücre (sayı) dönen sorgularda kullanılır. Bize 0 veya 1 döner.

                    if (upperGroupExists == 0) // Eğer 0 döndüyse, öyle bir ana klasör yok demektir!
                    {
                        // Eğer kayıt yoksa işlemi anında iptal et ve hata dön
                        return NotFound($"ID'si {request.UpperGroupId} olan bir Ana Klasör (Üst Grup) bulunamadı. Lütfen önce ana klasörü oluşturun.");
                    }
                }
                // --- GÜVENLİK POLİSİ ONAY VERDİ, KAYDA GEÇEBİLİRİZ ---

                string insertQuery = @"
                    INSERT INTO dbo.ImportGroup (UpperGroupId, Name, Code, IsActive, CreatedAt) 
                    VALUES (@UpperGroupId, @Name, @Code, 1, SYSUTCDATETIME())"; // SYSUTCDATETIME() o anki saat ve tarihi otomatik atar.

                using (var insertCommand = new SqlCommand(insertQuery, (SqlConnection)connection))
                {
                    // Dışarıdan gelen verileri SQL'e güvenli bir şekilde bağlıyoruz.
                    insertCommand.Parameters.AddWithValue("@UpperGroupId", request.UpperGroupId);
                    insertCommand.Parameters.AddWithValue("@Name", request.Name);
                    insertCommand.Parameters.AddWithValue("@Code", request.Code);

                    int affectedRows = insertCommand.ExecuteNonQuery(); // Kayıt işlemini tetikle. Bize etkilenen (eklenen) satır sayısını döner.

                    if (affectedRows > 0) // Eğer 0'dan büyükse kayıt başarıyla eklenmiştir.
                    {
                        return Ok("Yeni alt grup başarıyla ana klasöre bağlandı ve oluşturuldu."); // Başarı mesajı dön (HTTP 200).
                    }
                    else
                    {
                        return StatusCode(500, "Veritabanına kayıt sırasında bir hata oluştu."); // Beklenmedik bir sunucu hatası (HTTP 500).
                    }
                }
            }
        }
        [HttpDelete("{id}")]
        public IActionResult DeleteGroup(long id)
        {
            try
            {
                using (var connection = _dbService.CreateConnection())
                {
                    connection.Open();

                    // KORUMA: İçinde rapor var mı kontrol et
                    string checkQuery = "SELECT COUNT(*) FROM dbo.ImportReport WHERE GroupId = @Id";
                    using (var checkCommand = new SqlCommand(checkQuery, (SqlConnection)connection))
                    {
                        checkCommand.Parameters.AddWithValue("@Id", id);
                        int reportCount = (int)checkCommand.ExecuteScalar();

                        if (reportCount > 0)
                        {
                            return BadRequest($"Bu alt gruba bağlı {reportCount} adet rapor şablonu var. Lütfen önce o raporları silin.");
                        }
                    }

                    // İçinde rapor yoksa güvenle sil
                    string deleteQuery = "DELETE FROM dbo.ImportGroup WHERE Id = @Id";
                    using (var deleteCommand = new SqlCommand(deleteQuery, (SqlConnection)connection))
                    {
                        deleteCommand.Parameters.AddWithValue("@Id", id);
                        deleteCommand.ExecuteNonQuery();
                    }
                }
                return Ok(new { Message = "Alt Grup başarıyla silindi." });
            }
            catch (Exception ex)
            {
                return StatusCode(500, $"Hata: {ex.Message}");
            }
        }
    }

    // Dışarıdan gelen verinin şablonu (DTO) - Bu sefer 3 parametre istiyoruz
    public class CreateGroupRequest
    {
        public int UpperGroupId { get; set; } // Hangi üst gruba bağlanacak?
        public string Name { get; set; } // Alt grubun adı ne?
        public string Code { get; set; } // Alt grubun kodu ne?
    }
}

        // 1. HTTP GET - Raporları Listeleme
        [HttpGet]
        public IActionResult GetReports([FromQuery] string? roleCode) // DİKKAT: string yanına ? eklendi
        {
            var reports = new List<object>();

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open();

                // Temel sorgumuz (RoleCode kolonunu da çekiyoruz)
                string query = "SELECT Id, GroupId, Name, Code, PhysicalTableName, SchemaVersion, RoleCode FROM dbo.ImportReport WHERE IsActive = 1";

                // Eğer roleCode dolu gelmişse, sorguya filtre ekle (Kullanıcı Ekranı)
                // Eğer boş gelmişse filtre ekleme, tüm raporları getir (Yönetici Ekranı)
                if (!string.IsNullOrWhiteSpace(roleCode))
                {
                    query += " AND RoleCode = @RoleCode";
                }

                using (var command = new SqlCommand(query, (SqlConnection)connection))
                {
                    if (!string.IsNullOrWhiteSpace(roleCode))
                    {
                        command.Parameters.AddWithValue("@RoleCode", roleCode);
                    }

                    using (var reader = command.ExecuteReader())
                    {
                        while (reader.Read())
                        {
                            reports.Add(new
                            {
                                Id = Convert.ToInt64(reader["Id"]),
                                GroupId = Convert.ToInt64(reader["GroupId"]),
                                Name = reader["Name"]?.ToString(),
                                Code = reader["Code"]?.ToString(),
                                PhysicalTableName = reader["PhysicalTableName"] != DBNull.Value ? reader["PhysicalTableName"].ToString() : null,
                                SchemaVersion = Convert.ToInt32(reader["SchemaVersion"]),
                                RoleCode = reader["RoleCode"] != DBNull.Value ? reader["RoleCode"].ToString() : null
                            });
                        }
                    }
                }
            }
            return Ok(reports);
        }

        // 2. HTTP POST - Sisteme Yeni Rapor Ekleme
        [HttpPost]
        public IActionResult CreateReport([FromBody] CreateReportRequest request)
        {
            if (string.IsNullOrWhiteSpace(request.Name) || string.IsNullOrWhiteSpace(request.Code))
            {
                return BadRequest("Rapor adı ve kodu boş olamaz!");
            }

            if (request.GroupId <= 0)
            {
                return BadRequest("Geçerli bir Grup ID'si girmelisiniz!");
            }

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open();

                string checkQuery = @"
                        SELECT g.Code AS GroupCode, ug.Code AS UpperGroupCode 
                        FROM dbo.ImportGroup g
                        INNER JOIN dbo.ImportUpperGroup ug ON g.UpperGroupId = ug.Id
                        WHERE g.Id = @GroupId AND g.IsActive = 1";

                string groupCode = "";
                string upperGroupCode = "";

                using (var checkCommand = new SqlCommand(checkQuery, (SqlConnection)connection))
                {
                    checkCommand.Parameters.AddWithValue("@GroupId", request.GroupId);

                    using (var reader = checkCommand.ExecuteReader())
                    {
                        if (reader.Read())
                        {
                            groupCode = reader.GetString(0);
                            upperGroupCode = reader.GetString(1);
                        }
                        else
                        {
                            return NotFound($"ID'si {request.GroupId} olan aktif bir Alt Grup bulunamadı.");
                        }
                    }
                }

                string physicalTableName = $"rpt_{ToSafeSqlName(upperGroupCode)}_{ToSafeSqlName(groupCode)}_{ToSafeSqlName(request.Code)}";

                string insertQuery = @"
                        INSERT INTO dbo.ImportReport (GroupId, Name, Code, PhysicalTableName, IsActive, SchemaVersion, CreatedAt, RoleCode) 
                        VALUES (@GroupId, @Name, @Code, @PhysicalTableName, 1, 1, SYSDATETIME(), @RoleCode)";

                using (var insertCommand = new SqlCommand(insertQuery, (SqlConnection)connection))
                {
                    insertCommand.Parameters.AddWithValue("@GroupId", request.GroupId);
                    insertCommand.Parameters.AddWithValue("@Name", request.Name);
                    insertCommand.Parameters.AddWithValue("@Code", request.Code);
                    insertCommand.Parameters.AddWithValue("@PhysicalTableName", physicalTableName);

                    // YENİ EKLENDİ: Gelen Rol kodunu veritabanına bağlıyoruz
                    insertCommand.Parameters.AddWithValue("@RoleCode", string.IsNullOrEmpty(request.RoleCode) ? DBNull.Value : request.RoleCode);

                    int affectedRows = insertCommand.ExecuteNonQuery();

                    if (affectedRows > 0)
                    {
                        return Ok(new
                        {
                            Message = "Yeni rapor başarıyla oluşturuldu.",
                            GeneratedTableName = physicalTableName
                        });
                    }
                    else
                    {
                        return StatusCode(500, "Veritabanına rapor kaydı sırasında bir hata oluştu.");
                    }
                }
            }
        }

        // 3. HTTP GET - Dinamik Excel Şablonu İndirme
        [HttpGet("{id}/template")]
        public IActionResult DownloadTemplate(int id)
        {
            var fields = new List<CreateFieldRequest>();
            string reportCode = "";
            int schemaVersion = 1;

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open();

                // 1. Önce Raporun Kodu ve Versiyonunu alalım 
                string getReportQuery = "SELECT Code, SchemaVersion FROM dbo.ImportReport WHERE Id = @ReportId AND IsActive = 1";
                using (var cmd = new SqlCommand(getReportQuery, (SqlConnection)connection))
                {
                    cmd.Parameters.AddWithValue("@ReportId", id);
                    using (var reader = cmd.ExecuteReader())
                    {
                        if (reader.Read())
                        {
                            reportCode = reader.GetString(0);
                            schemaVersion = reader.GetInt32(1);
                        }
                        else
                        {
                            return NotFound("Aktif rapor bulunamadı.");
                        }
                    }
                }

                // 2. Bu rapora ait AKTİF alanları SortOrder (Sıra No) değerine göre dizerek çekelim
                // 2. Bu rapora ait AKTİF alanları SortOrder (Sıra No) değerine göre dizerek çekelim
                string getFieldsQuery = @"
                    SELECT ExcelColumnName, IsRequired, DataType, DisplayName 
                    FROM dbo.ImportReportField 
                    WHERE ReportId = @ReportId AND IsActive = 1 
                    ORDER BY SortOrder ASC";

                using (var cmd = new SqlCommand(getFieldsQuery, (SqlConnection)connection))
                {
                    cmd.Parameters.AddWithValue("@ReportId", id);
                    using (var reader = cmd.ExecuteReader())
                    {
                        while (reader.Read())
                        {
                            fields.Add(new CreateFieldRequest
                            {
                                ExcelColumnName = reader.GetString(0),
                                IsRequired = reader.GetBoolean(1),
                                DataType = reader.GetString(2),
                                DisplayName = reader.GetString(3) // Name yerine DisplayName kullanıyoruz
                            });
                        }
                    }
                }
            }

            if (fields.Count == 0)
            {
                return BadRequest("Bu rapora ait tanımlanmış aktif bir alan bulunmuyor. Lütfen önce alan listesi tanımlayın.");
            }

            // 3. CLOSEDXML İLE EXCEL DOSYASI OLUŞTURMA
            using (var workbook = new XLWorkbook())
            {
                var worksheet = workbook.Worksheets.Add("Veri_Girisi");

                for (int i = 0; i < fields.Count; i++)
                {
                    int colIndex = i + 1;
                    var cell = worksheet.Cell(1, colIndex);

                    cell.Value = fields[i].ExcelColumnName;
                    cell.Style.Font.Bold = true;
                    cell.Style.Fill.BackgroundColor = XLColor.LightGray;

                    string zorunluMetni = fields[i].IsRequired ? "ZORUNLU" : "Opsiyonel";
                    cell.GetComment().AddText($"Veri Tipi: {fields[i].DataType}\nDurum: {zorunluMetni}");
                }

                worksheet.Columns().AdjustToContents();

                // 4. DOSYAYI KULLANICIYA İNDİRTME
                using (var stream = new MemoryStream())
                {
                    workbook.SaveAs(stream);
                    var content = stream.ToArray();

                    string fileName = $"{reportCode}_sablon_v{schemaVersion}.xlsx";
                    string contentType = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";

                    return File(content, contentType, fileName);
                }
            }

        }

        // 4. HTTP POST - Excel Dosyası Yükleme (ANA METOT - CLEAN ARCHITECTURE)
        [HttpPost("{id}/imports")]
        public IActionResult ImportExcel(long id, IFormFile file)
        {
            if (file == null || file.Length == 0) return BadRequest("Lütfen yüklenecek bir Excel dosyası seçin.");
            if (Path.GetExtension(file.FileName).ToLowerInvariant() != ".xlsx") return BadRequest("Sadece .xlsx uzantılı dosyalar yüklenebilir.");

            long batchId = CreateImportBatchRecord(id, file.FileName, "STARTED");

            // 2. Adım: Veritabanından aktif alan (kolon) kurallarını getir
            var fields = GetActiveReportFields(id);
            if (fields.Count == 0)
            {
                UpdateImportBatchStatus(batchId, "FAILED", 0, 0, 0);
                return BadRequest("Bu rapora ait aktif alan tanımı bulunamadı.");
            }

            // 3. Adım: Excel'i oku ve kurallara göre doğrula
            var (parsedRows, errorLog) = ParseAndValidateExcel(file, fields);

            // 4. Adım: Hata varsa logla ve işlemi iptal et
            if (errorLog.Any())
            {
                LogErrorDetails(batchId, errorLog);
                UpdateImportBatchStatus(batchId, "FAILED_VALIDATION", parsedRows.Count + errorLog.Count, 0, errorLog.Count);

                var displayErrors = errorLog.Select(e => $"Satır {e.RowNumber} [{e.ExcelColumnName}]: {e.ErrorMessage}").ToList();

                return BadRequest(new
                {
                    BatchId = batchId,
                    Message = "Excel dosyasında verisel hatalar bulundu. İçeri aktarım durduruldu.",
                    Errors = displayErrors
                });
            }

            // 5. Adım: Hata yoksa Full Refresh (Eski veriyi sil ve yenileri ekle)
            try
            {
                // Veritabanı kayıt işlemini başlat
                ExecuteFullRefreshAndInsert(id, batchId, fields, parsedRows);

                // Hata fırlatmazsa işlem başarılı demektir
                UpdateImportBatchStatus(batchId, "SUCCESS", parsedRows.Count, parsedRows.Count, 0);
                return Ok(new { Message = $"{parsedRows.Count} satır veri başarıyla kaydedildi." });
            }
            catch (Exception ex)
            {
                // SQL'den gelen GERÇEK hatayı burada yakalıyoruz
                UpdateImportBatchStatus(batchId, "SYSTEM_ERROR", 0, 0, 0);
                return StatusCode(500, $"SQL Hatası: {ex.InnerException?.Message ?? ex.Message}");
            }
        }

        private (List<Dictionary<string, object>>, List<(int RowNumber, string ExcelColumnName, string FieldName, string RawValue, string ErrorMessage)>) ParseAndValidateExcel(IFormFile file, List<CreateFieldRequest> fields)
        {
            System.Text.Encoding.RegisterProvider(System.Text.CodePagesEncodingProvider.Instance);
            var errorLog = new List<(int, string, string, string, string)>();
            var parsedRows = new List<Dictionary<string, object>>();

            using (var stream = file.OpenReadStream())
            using (var reader = ExcelReaderFactory.CreateReader(stream))
            {
                if (!reader.Read()) return (parsedRows, errorLog);

                var headerIndices = new Dictionary<string, int>();
                for (int i = 0; i < reader.FieldCount; i++)
                {
                    var headerValue = reader.GetValue(i)?.ToString()?.Trim();
                    if (!string.IsNullOrEmpty(headerValue)) headerIndices[headerValue] = i;
                }

                int rowNumber = 2;
                while (reader.Read())
                {
                    var rowData = new Dictionary<string, object>();
                    bool isRowEmpty = true;

                    for (int i = 0; i < reader.FieldCount; i++)
                    {
                        if (reader.GetValue(i) != null && !string.IsNullOrWhiteSpace(reader.GetValue(i).ToString()))
                        {
                            isRowEmpty = false; break;
                        }
                    }
                    if (isRowEmpty) { rowNumber++; continue; }

                    foreach (var field in fields)
                    {
                        if (!headerIndices.ContainsKey(field.ExcelColumnName)) continue;

                        int colIndex = headerIndices[field.ExcelColumnName];
                        var rawValue = reader.GetValue(colIndex);
                        string stringValue = rawValue?.ToString()?.Trim() ?? "";

                        // 1. ZORUNLULUK KONTROLÜ
                        if (field.IsRequired && string.IsNullOrEmpty(stringValue))
                        {
                            errorLog.Add((rowNumber, field.ExcelColumnName, field.SqlColumnName, stringValue, "Bu alan zorunludur ve boş bırakılamaz."));
                            continue;
                        }

                        if (!string.IsNullOrEmpty(stringValue))
                        {
                            // 2. MAKSİMUM UZUNLUK KONTROLÜ (Artık "string" olarak arıyoruz)
                            if (field.DataType.ToLower() == "string" && field.MaxLength.HasValue && stringValue.Length > field.MaxLength.Value)
                            {
                                errorLog.Add((rowNumber, field.ExcelColumnName, field.SqlColumnName, stringValue, $"Metin en fazla {field.MaxLength.Value} karakter olmalıdır."));
                                continue;
                            }

                            // 3. VERİ TİPİ DÖNÜŞÜM VE DOĞRULAMA KONTROLÜ
                            try
                            {
                                object parsedValue = field.DataType.ToLower() switch
                                {
                                    "int" => Convert.ToInt64(rawValue),
                                    "decimal" => Convert.ToDecimal(rawValue),
                                    "datetime" => Convert.ToDateTime(rawValue),
                                    "boolean" => Convert.ToBoolean(rawValue),
                                    _ => stringValue
                                };
                                rowData[field.SqlColumnName] = parsedValue;
                            }
                            catch
                            {
                                // Dönüşüm başarısız olursa anında hata loguna at
                                errorLog.Add((rowNumber, field.ExcelColumnName, field.SqlColumnName, stringValue, $"Geçersiz veri formatı. Beklenen tip: {field.DataType.ToUpper()}."));
                            }
                        }
                        else
                        {
                            rowData[field.SqlColumnName] = DBNull.Value;
                        }
                    }
                    parsedRows.Add(rowData);
                    rowNumber++;
                }
            }
            return (parsedRows, errorLog);
        }

        private bool ExecuteFullRefreshAndInsert(long reportId, long batchId, List<CreateFieldRequest> fields, List<Dictionary<string, object>> parsedRows)
        {
            using (var conn = _dbService.CreateConnection())
            {
                conn.Open();
                using (var transaction = ((SqlConnection)conn).BeginTransaction())
                {
                    try
                    {
                        string tableName = "";
                        using (var cmd = new SqlCommand("SELECT PhysicalTableName FROM dbo.ImportReport WHERE Id = @ReportId", (SqlConnection)conn, transaction))
                        {
                            cmd.Parameters.AddWithValue("@ReportId", reportId);
                            tableName = cmd.ExecuteScalar()?.ToString();
                        }

                        if (string.IsNullOrEmpty(tableName)) throw new Exception("Tablo adı bulunamadı.");

                        // FULL REFRESH: Tablodaki eski veriyi sil
                        using (var delCmd = new SqlCommand($"DELETE FROM dbo.{tableName}", (SqlConnection)conn, transaction))
                        {
                            delCmd.ExecuteNonQuery();
                        }

                        // YENİ VERİLERİ EKLE
                        var sqlColumns = fields.Select(f => $"[{f.SqlColumnName}]").ToList();
                        var sqlParameters = fields.Select(f => $"@{f.SqlColumnName}").ToList();
                        string insertQuery = $"INSERT INTO dbo.{tableName} (ImportBatchId, RowNumber, {string.Join(", ", sqlColumns)}) VALUES (@ImportBatchId, @RowNumber, {string.Join(", ", sqlParameters)})";

                        int currentRowNum = 2;
                        foreach (var row in parsedRows)
                        {
                            using (var insertCmd = new SqlCommand(insertQuery, (SqlConnection)conn, transaction))
                            {
                                insertCmd.Parameters.AddWithValue("@ImportBatchId", batchId);
                                insertCmd.Parameters.AddWithValue("@RowNumber", currentRowNum);
                                foreach (var field in fields)
                                {
                                    insertCmd.Parameters.AddWithValue("@" + field.SqlColumnName, row.ContainsKey(field.SqlColumnName) ? row[field.SqlColumnName] : DBNull.Value);
                                }
                                insertCmd.ExecuteNonQuery();
                            }
                            currentRowNum++;
                        }
                        transaction.Commit();
                        return true;
                    }
                    catch (Exception ex)
                    {
                        transaction.Rollback();
                        throw;
                    }
                }
            }
        }

        [HttpGet("Data/{reportId}")]
        public IActionResult GetReportData(long reportId, [FromQuery] int pageNumber = 1, [FromQuery] int pageSize = 50)
        {
            try
            {
                var dataList = new List<Dictionary<string, object>>();
                int totalRecords = 0;

                using (var connection = _dbService.CreateConnection())
                {
                    connection.Open();

                    // 1. ADIM: Raporun fiziksel tablo adını bulalım
                    string getTableQuery = "SELECT PhysicalTableName FROM dbo.ImportReport WHERE Id = @ReportId";
                    string physicalTableName = "";

                    using (var getTableCmd = new SqlCommand(getTableQuery, (SqlConnection)connection))
                    {
                        getTableCmd.Parameters.AddWithValue("@ReportId", reportId);
                        var result = getTableCmd.ExecuteScalar();
                        if (result == null) return NotFound("Rapor bulunamadı.");

                        physicalTableName = result.ToString().ToLowerInvariant();
                    }

                    // 2. ADIM: Tablodaki toplam kayıt sayısını (Total Count) alalım - Paging için şart
                    string countQuery = $"SELECT COUNT(1) FROM dbo.[{physicalTableName}]";
                    using (var countCmd = new SqlCommand(countQuery, (SqlConnection)connection))
                    {
                        totalRecords = (int)countCmd.ExecuteScalar();
                    }

                    // 3. ADIM: Dinamik SQL ile Sayfalama (Paging) yaparak verileri çekelim
                    int offset = (pageNumber - 1) * pageSize;
                    string dataQuery = $@"
                SELECT * FROM dbo.[{physicalTableName}] 
                ORDER BY Id 
                OFFSET @Offset ROWS 
                FETCH NEXT @PageSize ROWS ONLY";

                    using (var dataCmd = new SqlCommand(dataQuery, (SqlConnection)connection))
                    {
                        dataCmd.Parameters.AddWithValue("@Offset", offset);
                        dataCmd.Parameters.AddWithValue("@PageSize", pageSize);

                        using (var reader = dataCmd.ExecuteReader())
                        {
                            while (reader.Read())
                            {
                                var row = new Dictionary<string, object>();
                                for (int i = 0; i < reader.FieldCount; i++)
                                {
                                    if (reader.IsDBNull(i))
                                    {
                                        row[reader.GetName(i)] = null;
                                    }
                                    else
                                    {
                                        var value = reader.GetValue(i);

                                        // Eğer gelen veri bir Tarih (DateTime) ise formatla
                                        if (value is DateTime dt)
                                        {
                                            // Eğer saati 00:00:00 ise sadece tarihi göster (Örn: AsimTarihi)
                                            if (dt.TimeOfDay == TimeSpan.Zero)
                                            {
                                                row[reader.GetName(i)] = dt.ToString("dd.MM.yyyy");
                                            }
                                            // Eğer içinde saat bilgisi de varsa saati de güzel bir formatta göster (Örn: CreatedAt)
                                            else
                                            {
                                                row[reader.GetName(i)] = dt.ToString("dd.MM.yyyy HH:mm:ss");
                                            }
                                        }
                                        else
                                        {
                                            // Tarih değilse veriyi olduğu gibi bırak
                                            row[reader.GetName(i)] = value;
                                        }
                                    }
                                }
                                dataList.Add(row);
                            }
                        }
                    }
                }

                // Toplam kayıt sayısını ve sayfalanmış veriyi Blazor'a dönüyoruz
                return Ok(new
                {
                    TotalRecords = totalRecords,
                    PageNumber = pageNumber,
                    PageSize = pageSize,
                    Data = dataList
                });
            }
            catch (Exception ex)
            {
                return StatusCode(500, $"Veriler listelenirken hata oluştu: {ex.Message}");
            }
        }

        private string ToSafeSqlName(string input)
        {
            if (string.IsNullOrWhiteSpace(input)) return string.Empty;

            return input.Trim().ToLowerInvariant()
                .Replace("ı", "i")
                .Replace("ğ", "g")
                .Replace("ü", "u")
                .Replace("ş", "s")
                .Replace("ö", "o")
                .Replace("ç", "c")
                .Replace(" ", "_")
                .Replace("-", "_");
        }
    }

        // 2. HTTP POST - Sisteme Yeni Veri Ekleme
        [HttpPost]
        public IActionResult CreateUpperGroup([FromBody] CreateUpperGroupRequest request)
        {
            if (string.IsNullOrWhiteSpace(request.Name) || string.IsNullOrWhiteSpace(request.Code))
                return BadRequest("Grup adı ve kodu boş olamaz!");

            using (var connection = _dbService.CreateConnection())
            {
                connection.Open();
                string query = @"
                INSERT INTO dbo.ImportUpperGroup (Name, Code, IsActive, CreatedAt, UpdatedAt) 
                VALUES (@Name, @Code, 1, SYSUTCDATETIME(), SYSUTCDATETIME())";

                using (var command = new SqlCommand(query, (SqlConnection)connection))
                {
                    command.Parameters.AddWithValue("@Name", request.Name);
                    command.Parameters.AddWithValue("@Code", request.Code);

                    int affectedRows = command.ExecuteNonQuery();
                    if (affectedRows > 0) return Ok("Yeni üst grup başarıyla oluşturuldu.");
                    else return StatusCode(500, "Veritabanına kayıt sırasında bir hata oluştu.");
                }
            }
        }
        [HttpDelete("{id}")]
        public IActionResult DeleteUpperGroup(long id)
        {
            try
            {
                using (var connection = _dbService.CreateConnection())
                {
                    connection.Open();

                    // KORUMA: İçinde alt grup var mı kontrol et
                    string checkQuery = "SELECT COUNT(*) FROM dbo.ImportGroup WHERE UpperGroupId = @Id";
                    using (var checkCommand = new SqlCommand(checkQuery, (SqlConnection)connection))
                    {
                        checkCommand.Parameters.AddWithValue("@Id", id);
                        int groupCount = (int)checkCommand.ExecuteScalar();

                        if (groupCount > 0)
                        {
                            return BadRequest($"Bu üst gruba bağlı {groupCount} adet alt grup var. Lütfen önce o alt grupları silin.");
                        }
                    }

                    // İçinde alt grup yoksa güvenle sil
                    string deleteQuery = "DELETE FROM dbo.ImportUpperGroup WHERE Id = @Id";
                    using (var deleteCommand = new SqlCommand(deleteQuery, (SqlConnection)connection))
                    {
                        deleteCommand.Parameters.AddWithValue("@Id", id);
                        deleteCommand.ExecuteNonQuery();
                    }
                }
                return Ok(new { Message = "Üst Grup başarıyla silindi." });
            }
            catch (Exception ex)
            {
                return StatusCode(500, $"Hata: {ex.Message}");
            }
        }
    }

	private int totalPages => totalRecords > 0 ? (int)Math.Ceiling((double)totalRecords / pageSize) : 1;

	// SSO'dan geldiğini varsaydığımız test rolü. Gerçek projede bu değer Login olan kullanıcının Token'ından okunur.
	private string currentUserRole = "1000";

	private async Task DownloadTemplate()
	{
		try
		{
			successMessage = ""; errorMessage = "";
			var downloadUrl = $"{Http.BaseAddress}api/ImportReport/{selectedReportId}/template";
			await JSRuntime.InvokeVoidAsync("open", downloadUrl, "_blank");
		}

	private void HandleFileSelected(InputFileChangeEventArgs e)
	{
		selectedFile = e.File;
		successMessage = ""; errorMessage = ""; validationErrors.Clear(); errorBatchId = null;
	}

	// --- DOSYAYI API'YE GÖNDEREN FONKSİYON ---
	private async Task UploadExcelFile()
	{
		if (selectedFile == null) return;

		isProcessing = true;
		successMessage = ""; errorMessage = ""; validationErrors.Clear(); errorBatchId = null;

		try
		{
			using var content = new MultipartFormDataContent();
			var fileStream = selectedFile.OpenReadStream(maxAllowedSize: 10 * 1024 * 1024);
			var streamContent = new StreamContent(fileStream);
			content.Add(streamContent, "file", selectedFile.Name);

			var response = await Http.PostAsync($"api/ImportReport/{selectedReportId}/imports", content);

			if (response.IsSuccessStatusCode)
			{
				successMessage = $"'{selectedFile.Name}' başarıyla yüklendi! Eski veriler silindi ve güncel veriler SQL tablolarına yerleştirildi (Full Refresh Başarılı).";
				selectedFile = null;

				// BAŞARILI YÜKLEMEDEN SONRA TABLOYU OTOMATİK YENİLE
				currentPage = 1;
				await LoadData();
			}
			else if (response.StatusCode == System.Net.HttpStatusCode.BadRequest)
			{
				var errorResult = await response.Content.ReadFromJsonAsync<ApiErrorResponse>(jsonOptions);
				if (errorResult != null)
				{
					errorMessage = errorResult.Message;
					validationErrors = errorResult.Errors ?? new();
					errorBatchId = errorResult.BatchId;
				}
			}
			else
			{
				string errorDetails = await response.Content.ReadAsStringAsync();
				errorMessage = $"Sunucu Hatası ({response.StatusCode}): {errorDetails}";
			}
		}
		catch (Exception ex)
		{
			errorMessage = $"Dosya sunucuya aktarılırken bir ağ veya sistem hatası oluştu: {ex.Message}";
		}
		finally
		{
			isProcessing = false;
		}
	}

	// --- VERİ LİSTELEME VE SAYFALAMA (PAGING) ---
	private async Task LoadData()
	{
		if (string.IsNullOrEmpty(selectedReportId)) return;

		var response = await Http.GetFromJsonAsync<PagedDataResponse>($"api/ImportReport/Data/{selectedReportId}?pageNumber={currentPage}&pageSize={pageSize}", jsonOptions);
		if (response != null)
		{
			dataList = response.Data;
			totalRecords = response.TotalRecords;
		}
	}

	private async Task NextPage() { currentPage++; await LoadData(); }
	private async Task PrevPage() { currentPage--; await LoadData(); }

	private void OnUpperGroupChanged(ChangeEventArgs e)

		successMessage = ""; errorMessage = ""; validationErrors.Clear(); errorBatchId = null;
		filteredGroups.Clear(); filteredReports.Clear(); dataList.Clear();

		if (!string.IsNullOrEmpty(selectedUpperGroupId))
		{
			long upperId = Convert.ToInt64(selectedUpperGroupId);
			filteredGroups = allGroups.Where(g => g.UpperGroupId == upperId).ToList();
		}
	}

	private void OnGroupChanged(ChangeEventArgs e)
	{
		selectedGroupId = e.Value?.ToString();
		selectedReportId = null; selectedFile = null;
		successMessage = ""; errorMessage = ""; validationErrors.Clear(); errorBatchId = null;
		filteredReports.Clear(); dataList.Clear();

		if (!string.IsNullOrEmpty(selectedGroupId))
		{
			long groupId = Convert.ToInt64(selectedGroupId);
			filteredReports = allReports.Where(r => r.GroupId == groupId).ToList();
		}
	}

	private void OnReportChanged(ChangeEventArgs e)
	{
		selectedReportId = e.Value?.ToString();
		selectedFile = null;
		successMessage = ""; errorMessage = ""; validationErrors.Clear(); errorBatchId = null;
		dataList.Clear();
	}
